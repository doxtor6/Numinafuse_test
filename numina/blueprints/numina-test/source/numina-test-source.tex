
\begin{definition}[Laplacian matrix]
\label{def:laplacian}
Let \(G=(V,E)\) be a simple undirected graph on \(n\) vertices.
Let \(A(G)\in\mathbb{R}^{n\times n}\) be its adjacency matrix and let
\(D(G)\in\mathbb{R}^{n\times n}\) be the diagonal degree matrix, whose
\(i\)-th diagonal entry equals \(\deg_G(i)\).
The \emph{Laplacian matrix} of \(G\) is
\[
L(G)=D(G)-A(G).
\]
\end{definition}

\begin{definition}[Algebraic connectivity]
\label{def:algebraic_connectivity}
\uses{def:laplacian}
The matrix \(L(G)\) is real symmetric and positive semidefinite, so its
eigenvalues are real and nonnegative.
We order them as
\[
0=\lambda_1(G)\le \lambda_2(G)\le \cdots \le \lambda_n(G).
\]
The second smallest eigenvalue \(\lambda_2(G)\) is called the
\emph{algebraic connectivity} of \(G\).
\end{definition}

\begin{definition}[Complete bipartite graph]
\label{def:complete_bipartite}
Let \(a,b\ge 1\).
The \emph{complete bipartite graph} \(K_{a,b}\) is the simple undirected
graph whose vertex set can be partitioned as
\[
V = V_1 \sqcup V_2, \quad |V_1|=a,\; |V_2|=b,
\]
and whose edge set consists of all pairs \(\{u,v\}\) with \(u\in V_1\)
and \(v\in V_2\); there are no edges within \(V_1\) or within \(V_2\).
\end{definition}

\begin{lemma}[Algebraic connectivity of \(K_{2,n-2}\)]
\label{lem:lambda2_k2_nm2}
\uses{def:algebraic_connectivity, def:complete_bipartite}
For every integer \(n\ge 4\),
\[
\lambda_2(K_{2,n-2}) = 2.
\]
\end{lemma}
\begin{proof}
\uses{def:laplacian, def:complete_bipartite}
Write \(m = n - 2\), so \(m \ge 2\), and fix a bipartition
\(V = V_1 \sqcup V_2\) of \(K_{2,m}\) with
\(V_1 = \{u_1, u_2\}\) and \(V_2 = \{v_1, \dots, v_m\}\).
Each vertex in \(V_1\) has degree \(m\) and each vertex in \(V_2\)
has degree \(2\), so with the vertex order
\(u_1, u_2, v_1, \dots, v_m\) the Laplacian \(L = D - A\) has the block
form
\[
L =
\begin{pmatrix}
m\, I_2 & -J_{2,m} \\
-J_{m,2} & 2\, I_m
\end{pmatrix},
\]
where \(J_{r,s}\) denotes the \(r \times s\) all-ones matrix.

We exhibit four families of eigenvectors with respective eigenvalues
\(0\), \(2\), \(m\), and \(m + 2\); their multiplicities add to \(n\),
which identifies the full spectrum.

\medskip
\noindent\emph{Eigenvalue \(0\), multiplicity \(1\).}\;
The all-ones vector \(\mathbf 1 \in \mathbb R^n\) satisfies
\(L \mathbf 1 = 0\), since every row of a Laplacian matrix sums to zero.

\medskip
\noindent\emph{Eigenvalue \(2\), multiplicity \(m - 1\).}\;
For any \((y_1, \dots, y_m) \in \mathbb R^m\) with
\(\sum_{i=1}^m y_i = 0\), set \(x = (0, 0, y_1, \dots, y_m)^T\). Then
\[
(Lx)_{u_j} = m \cdot 0 - \sum_{i=1}^m y_i = 0
\quad (j = 1, 2),
\qquad
(Lx)_{v_i} = 2 y_i - 0 - 0 = 2 y_i,
\]
so \(L x = 2 x\). The space of such \((y_i)\) has dimension \(m - 1\).

\medskip
\noindent\emph{Eigenvalue \(m\), multiplicity \(1\).}\;
For \(z = (1, -1, 0, \dots, 0)^T\),
\[
(Lz)_{u_1} = m,
\qquad
(Lz)_{u_2} = -m,
\qquad
(Lz)_{v_i} = 2 \cdot 0 - 1 - (-1) = 0,
\]
so \(L z = m z\).

\medskip
\noindent\emph{Eigenvalue \(m + 2\), multiplicity \(1\).}\;
For \(w = (m, m, -2, \dots, -2)^T\),
\[
(Lw)_{u_j} = m \cdot m - \sum_{i=1}^m (-2) = m(m + 2),
\qquad
(Lw)_{v_i} = 2 \cdot (-2) - m - m = -2(m + 2),
\]
so \(L w = (m + 2) w\). Note \(w\) is orthogonal to \(\mathbf 1\) and
not constant on \(V_1\)-and-\(V_2\) at the same value, so it is linearly
independent from the eigenvectors above.

\medskip
The total multiplicity is
\(1 + (m - 1) + 1 + 1 = m + 2 = n\), so these are all eigenvalues of
\(L\). Substituting \(m = n - 2\), the spectrum of \(L(K_{2, n-2})\) is
\[
\{\, 0,\; \underbrace{2, \dots, 2}_{n-3},\; n - 2,\; n \,\}.
\]
Since \(n \ge 4\) gives \(n - 2 \ge 2\), the sorted spectrum is
\(0 \le 2 \le 2 \le \dots \le n - 2 \le n\), so the second smallest
eigenvalue is \(2\). By the definition of algebraic connectivity,
\(\lambda_2(K_{2, n-2}) = 2\).
\end{proof}

\begin{theorem}[Edge-count bound on algebraic connectivity]
\label{thm:lambda2_bound_m_2nm2}
\uses{def:algebraic_connectivity}
Let \(n\ge 4\), and let \(G\) be a simple undirected graph on \(n\)
vertices with exactly
\[
|E(G)|=2(n-2)
\]
edges.
Then
\[
\lambda_2(G)\le 2.
\]
\end{theorem}

\begin{theorem}[\(K_{2,n-2}\) maximizes algebraic connectivity for \(m=2(n-2)\)]
\label{thm:k2nminus2_maximizer}
\uses{def:algebraic_connectivity, def:complete_bipartite,
      lem:lambda2_k2_nm2, thm:lambda2_bound_m_2nm2}
Let \(n\ge 4\).
By Theorem~\ref{thm:lambda2_bound_m_2nm2}, the complete
bipartite graph \(K_{2,n-2}\) is a maximizer of algebraic connectivity
among all simple undirected graphs on \(n\) vertices with exactly
\(2(n-2)\) edges:
\[
\lambda_2(K_{2,n-2}) = 2,
\qquad
\lambda_2(G) \le \lambda_2(K_{2,n-2})
\quad
\text{for every such } G.
\]
\end{theorem}
