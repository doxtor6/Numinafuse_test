\begin{theorem}[Colorful Carathéodory, Bárány 1982]
\label{thm:colorful_caratheodory}
\lean{NuminafuseTest.Blueprint.MainTheorem}
\leanfile{NuminafuseTest/Blueprint.lean}
Let \(d \in \mathbb{N}\) and let \(C_0, C_1, \dots, C_d\) be \(d+1\)
subsets of the Euclidean space \(\mathbb{R}^d\). Suppose that the
origin lies in the convex hull of each colour class:
\[
0 \in \operatorname{conv}(C_i) \qquad \text{for every } i \in \{0, 1, \dots, d\}.
\]
Then there exist points \(p_0, p_1, \dots, p_d\), one from each colour
class \(p_i \in C_i\), such that the origin lies in the convex hull of
the chosen rainbow simplex:
\[
0 \in \operatorname{conv}\{p_0, p_1, \dots, p_d\}.
\]
\end{theorem}
